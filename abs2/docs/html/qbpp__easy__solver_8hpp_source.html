79  void insert_if_better(const qbpp::Sol &sol) {
80  static std::mutex mutex_;
81  if (max_best_sol_count_ == 0)
82  return;
83  if (worst_energy_ < sol.energy())
84  return;
85  std::lock_guard<std::mutex> lock(mutex_);
86  if (sol_set_.count(sol))
87  return;
88  if (sol_vector_.size() < max_best_sol_count_) {
89  sol_vector_.push_back(sol);
90  sol_set_.insert(sol);
91  } else {
92  if (sol.energy() >= sol_vector_.back().energy())
93  return;
94  sol_set_.erase(sol_vector_.back());
95  sol_vector_.back() = sol;
96  sol_set_.insert(sol);
97  }
98 
99  for (size_t i = sol_vector_.size() - 1; i > 0; --i) {
100  if (sol_vector_[i].energy() < sol_vector_[i - 1].energy()) {
101  std::swap(sol_vector_[i], sol_vector_[i - 1]);
102  } else {
103  break;
104  }
105  }
106  worst_energy_ = sol_vector_.back().energy();
107  }

364 inline void SolDelta::flip(vindex_t index) {
365  if (index >= var_count()) {
366  throw std::out_of_range(
367  THROW_MESSAGE("Flip index (", index, ") is out of range."));
368  }
369 
370  for (vindex_t j = 0; j < quad_model_.degree(index); ++j) {
371  auto [k, coeff] = quad_model_.quadratic(index, j);
372 
373  if (delta_[k] < 0) {
374  neg_set_.erase(k);
375  }
376  before_delta_updated(k);
377 
378  delta_[k] +=
379  static_cast<energy_t>((2 * get(index) - 1) * (2 * get(k) - 1) * coeff);
380 
381  if (delta_[k] < 0) {
382  neg_set_.insert(k, delta_[k]);
383  }
384  after_delta_updated(k);
385  }
386 
387  if (delta_[index] < 0) {
388  neg_set_.erase(index);
389  }
390  before_delta_updated(index);
391 
392  delta_[index] = -delta_[index];
393 
394  if (delta_[index] < 0) {
395  neg_set_.insert(index, delta_[index]);
396  }
397 
398  after_delta_updated(index);
399 
400  flip_bit_add_delta(index, -delta_[index]);
401 
402  ++flip_count_;
403 }

421 inline void SolDelta::move_to(qbpp::Sol destination) {
422  misc::MinSet to_be_flipped;
423  for (vindex_t i = 0; i < var_count(); ++i) {
424  if (get(i) != destination.get(i)) {
425  to_be_flipped.insert(i, delta_[i]);
426  }
427  }
428  while (to_be_flipped.var_count() > 0) {
429  vindex_t min_index = to_be_flipped.get_first();
430  for (vindex_t j = 0; j < quad_model_.degree(min_index); ++j) {
431  auto pair = quad_model_.quadratic(min_index, j);
432  if (get(pair.first) != destination.get(pair.first)) {
433  to_be_flipped.erase(pair.first, delta_[pair.first]);
434  }
435  }
436  to_be_flipped.erase(min_index, delta_[min_index]);
437  flip(min_index);
438  for (vindex_t j = 0; j < quad_model_.degree(min_index); ++j) {
439  auto pair = quad_model_.quadratic(min_index, j);
440  if (get(pair.first) != destination.get(pair.first))
441  to_be_flipped.insert(pair.first, delta_[pair.first]);
442  }
443  set_if_better("Easy");
444  }

474 inline void PosMinSolDelta::search(size_t iteration) {
475  greedy();
476  for (vindex_t i = 0; i < iteration; ++i) {
477  std::optional<vindex_t> flip_index = std::nullopt;
478  for (vindex_t j = 0; j < tabu_size_ * 2; ++j) {
479  vindex_t candidate = vindex_limit;
480  if (pos_count() == 0 || neg_count() == 0) {
481  candidate = qbpp::misc::RandomGenerator::gen(var_count());
482  } else if (qbpp::misc::RandomGenerator::gen(neg_count() + 1) == 0) {
483  candidate = pos_min();
484  } else {
485  candidate = neg_random();
486  }
487  if (candidate == vindex_limit) {
488  throw std::runtime_error(THROW_MESSAGE("Unexpected error."));
489  }
490  if (!tabu_.has(candidate)) {
491  flip_index = candidate;
492  break;
493  }
494  }
495  if (!flip_index.has_value()) {
496  flip_index = tabu_.non_tabu_random();
497  }
498  flip(flip_index.value());
499  tabu_.insert(flip_index.value());
500  set_if_better("Easy");
501  if (easy_solver_.get_target_energy().has_value() &&
502  sol_holder_ptr_->energy() <= easy_solver_.get_target_energy())
503  break;
504  }
505  greedy();

508 inline void EasySolver::single_search(size_t thread_id) {
509  const size_t min_flip_count = 100;
510  PosMinSolDelta pos_min_sol_delta(*this);
511  size_t max_flip_count = (var_count() / 2);
512  if (thread_count_ > 1) {
513  double k = std::pow(static_cast<double>(max_flip_count) / min_flip_count,
514  1.0 / static_cast<double>(thread_count_ - 1));
515  max_flip_count =
516  static_cast<size_t>(min_flip_count * std::pow(k, thread_id));
517  if (max_flip_count < min_flip_count)
518  max_flip_count = min_flip_count;
519  }
520  energy_t prev_energy = sol_holder_ptr_->energy();
521  size_t random_flip_count = 1;
522  while (!time_limit_.has_value() ||
523  qbpp::get_time() - start_time_ < time_limit_.value()) {
524  pos_min_sol_delta.move_to(sol_holder_ptr_->get_sol());
525  pos_min_sol_delta.random_flip(random_flip_count);
526  pos_min_sol_delta.search(max_flip_count);
527  if (target_energy_.has_value() &&
528  sol_holder_ptr_->energy() <= target_energy_.value())
529  break;
530  if (prev_energy == sol_holder_ptr_->energy()) {
531  random_flip_count = (random_flip_count + 1) % (max_flip_count / 2);
532  if (random_flip_count == 0)
533  random_flip_count = 1;
534  } else {
535  random_flip_count = 1;
536  }
537  prev_energy = sol_holder_ptr_->energy();
538  }
539 
540  {
541  std::lock_guard<std::mutex> lock(flip_count_mutex_);
542  flip_count_ += pos_min_sol_delta.get_flip_count();
543  }
544 }
