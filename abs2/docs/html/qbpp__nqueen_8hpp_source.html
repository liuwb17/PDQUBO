38  static qbpp::QuadModel expand_mode(int dim,
39  const Vector<Vector<qbpp::Var>> &X) {
40  // Create arrays of expressions to store the sums of the diagonals.
41 
42  auto a = qbpp::expr(2 * dim - 3);
43  auto b = qbpp::expr(2 * dim - 3);
44  for (int i = 0; i < 2 * dim - 3; i++) {
45  int k = i + 1;
46  for (int j = 0; j < dim; j++) {
47  if (k - j >= 0 && k - j < dim) {
48  a[i] += X[j][k - j];
49  }
50  }
51  int l = dim - i - 1;
52  for (int j = 0; j < dim; j++) {
53  if (l + j >= 0 && l + j < dim) {
54  b[i] += X[j][l + j];
55  }
56  }
57  }
58 
59  auto expr = qbpp::sum(qbpp::vector_sum(X) == 1);
60  expr += qbpp::sum(qbpp::vector_sum(qbpp::transpose(X)) == 1);
61  expr += qbpp::sum(0 <= a <= 1);
62  expr += qbpp::sum(0 <= b <= 1);
63  expr.simplify_as_binary();
64 
65  return qbpp::QuadModel(expr);
66  }

73  static qbpp::QuadModel fast_mode(int dim,
74  const Vector<Vector<qbpp::Var>> &X) {
75  qbpp::Expr expr = dim;
76  for (int x = 0; x < dim; ++x) {
77  for (int y = 0; y < dim; ++y) {
78  // Reward for placing a queen
79  expr -= X[x][y];
80  for (int i = x + 1; i < dim; ++i) {
81  // Penalty for placing two queens in the same row
82  expr += X[x][y] * X[i][y];
83  }
84  for (int j = y + 1; j < dim; ++j) {
85  // Penalty for placing two queens in the same column
86  expr += X[x][y] * X[x][j];
87  }
88  for (int d = 1; d < dim - x; ++d) {
89  if (y - d >= 0) {
90  // Penalty for placing two queens on the same anti-diagonal
91  expr += X[x][y] * X[x + d][y - d];
92  }
93  if (y + d < dim) {
94  // Penalty for placing two queens on the same diagonal
95  expr += X[x][y] * X[x + d][y + d];
96  }
97  }
98  }
99  }
100  return qbpp::QuadModel(expr.simplify_as_binary());

116  for (int j = y + 1; j < dim; ++j) {
