323 inline void TSPMap::gen_random_map(uint32_t n) {
324  nodes.reserve(n);
325  uint32_t x, y;
326  for (uint32_t i = 0; i < n; i++) {
327  uint32_t counter = 0;
328  uint32_t max_dist = 1;
329  // Terminate if dist>=max_dist is satisfied 10 times.
330  while (counter < 10) {
331  x = qbpp::misc::RandomGenerator::gen(grid_size);
332  y = qbpp::misc::RandomGenerator::gen(grid_size);
333  uint32_t dist = min_dist(x, y);
334  if (dist >= max_dist) {
335  max_dist = dist;
336  counter++;
337  }
338  }
339  add_node(x, y);
340  }
341 }

347 inline std::tuple<qbpp::Model, bool, qbpp::Vector<qbpp::Vector<qbpp::Var>>>
348 TSPQuadModel::helper_func(const TSPMap &tsp_map, bool fix_first) {
349  auto node_count = tsp_map.node_count();
350  auto x = qbpp::var("x", node_count, node_count);
351  std::cout << "Generating QUBO expression for permutation." << std::endl;
352  auto permutation_expr = qbpp::sum(
353  (qbpp::vector_sum(x) == 1) + (qbpp::vector_sum(qbpp::transpose(x)) == 1));
354  std::cout << "Generating QUBO expressions for tour distances." << std::endl;
355 
356  qbpp::Vector<qbpp::Expr> exprs(node_count);
357  tbb::parallel_for(decltype(node_count)(0), node_count, [&](int i) {
358  uint32_t next_i = (i + 1) % node_count;
359  auto &expr = exprs[i];
360 
361  for (uint32_t j = 0; j < node_count; j++) {
362  for (uint32_t k = 0; k < node_count; k++) {
363  if (j == k)
364  continue;
365  expr += tsp_map.dist(j, k) * x[i][j] * x[next_i][k];
366  }
367  }
368  });
369 
370  auto tsp_expr = qbpp::sum(exprs) + permutation_expr * tsp_map.get_grid_size();
371 
372  if (fix_first) {
373  std::cout << "Fixing the first visiting node." << std::endl;
374  qbpp::MapList fix0 = {{x[0][0], 1}};
375  for (uint32_t i = 1; i < node_count; i++) {
376  fix0.push_back({x[0][i], 0});
377  fix0.push_back({x[i][0], 0});
378  }
379  tsp_expr.replace(fix0);
380  }
381 
382  std::cout << "Simplifying the QUBO expression." << std::endl;
383 
384  tsp_expr.simplify_as_binary();
385 
386  return {tsp_expr, fix_first, x};
387 }

393 inline std::vector<uint32_t>
394 TSPSol::gen_tour(const TSPQuadModel &tsp_quad_model, const Sol &sol) {
395  std::vector<uint32_t> tour;
396  for (uint32_t i = 0; i < tsp_quad_model.node_count(); i++) {
397  if (tsp_quad_model.get_fix_first() && i == 0) {
398  tour.push_back(0);
399  continue;
400  }
401  uint32_t count = 0;
402  uint32_t node;
403  for (uint32_t j = (tsp_quad_model.get_fix_first() ? 1 : 0);
404  j < tsp_quad_model.node_count(); j++) {
405  if (sol.get(tsp_quad_model.get_var(i, j)) == 1) {
406  node = j;
407  count++;
408  }
409  }
410  if (count != 1)
411  node = uint32_limit;
412  tour.push_back(node);
413  }
414  return tour;
415 }
