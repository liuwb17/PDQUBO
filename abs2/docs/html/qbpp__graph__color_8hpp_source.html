199 inline void GraphColorMap::gen_random_map(uint32_t n, bool is_circle) {
200  nodes_.reserve(n);
201  uint32_t x, y;
202  for (uint32_t i = 0; i < n; i++) {
203  uint32_t counter = 0;
204  uint32_t max_dist = 1;
205  // Terminate if dist>=max_dist is satisfied 10 times.
206  while (counter < 10) {
207  if (is_circle) {
208  do {
209  x = qbpp::misc::RandomGenerator::gen(grid_size_);
210  y = qbpp::misc::RandomGenerator::gen(grid_size_);
211  } while ((x - grid_size_ / 2) * (x - grid_size_ / 2) +
212  (y - grid_size_ / 2) * (y - grid_size_ / 2) >
213  grid_size_ * grid_size_ / 4);
214  } else {
215  x = qbpp::misc::RandomGenerator::gen(grid_size_);
216  y = qbpp::misc::RandomGenerator::gen(grid_size_);
217  }
218  uint32_t dist = min_dist(x, y);
219  if (dist >= max_dist) {
220  max_dist = dist;
221  counter++;
222  }
223  }
224  add_node(x, y);
225  }
226 }

238 inline void GraphColorMap::gen_delaunay_edges() {
239  typedef boost::polygon::point_data<int> Point;
240  std::vector<Point> points;
241  for (const auto &[x, y] : nodes_) {
242  points.push_back(Point(x, y));
243  }
244  boost::polygon::voronoi_diagram<double> vd;
245  boost::polygon::construct_voronoi(points.begin(), points.end(), &vd);
246  for (const auto &cell : vd.cells()) {
247  size_t source_index = cell.source_index();
248  const auto *edge = cell.incident_edge();
249  do {
250  size_t twin_source_index = edge->twin()->cell()->source_index();
251  if (source_index < twin_source_index)
252  edges_.push_back({static_cast<uint32_t>(source_index),
253  static_cast<uint32_t>(twin_source_index)});
254  edge = edge->next();
255  } while (edge != cell.incident_edge());
256  }

359 inline std::pair<qbpp::Model, qbpp::Vector<qbpp::Vector<qbpp::Var>>>
360 GraphColorQuadModel::helper_func(const GraphColorMap &graph_map,
361  uint32_t color_count) {
362  auto x = qbpp::var("x", graph_map.node_count(), color_count);
363 
364  auto f = qbpp::sum(qbpp::vector_sum(x) == 1);
365 
366  for (auto [i, j] : graph_map.get_edges()) {
367  f += qbpp::sum(x[i] * x[j]);
368  }
369  return {simplify_as_binary(f), x};
370 }
