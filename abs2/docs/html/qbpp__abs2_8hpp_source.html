251 inline QuadModel::QuadModel(const qbpp::QuadModel &quad_model)
252  : qbpp::QuadModel(quad_model) {
253  // Create an empty ABS2 model
254  abs2model_ptr = std::make_unique<abs2::Model>(
255  var_count(), quad_model.min_coeff(), quad_model.max_coeff());
256 
257  // Set the linear and quadratic terms
258  for (qbpp::vindex_t i = 0; i < var_count(); i++) {
259  if (quad_model.linear(i) != 0)
260  abs2model_ptr->set(i, i, quad_model.linear(i));
261  }
262  for (qbpp::vindex_t i = 0; i < var_count(); i++) {
263  for (qbpp::vindex_t j = 0; j < quad_model.degree(i); j++) {
264  auto [k, coeff] = quad_model.quadratic(i, j);
265  abs2model_ptr->set(i, k, coeff);
266  }
267  }
268 }

274 // Execute ABS2 for "quad_model" with "param" and returns "sol".
275 inline Sol Solver::operator()(const QuadModel &quad_model, Param &param) const {
276  // target_energy is adjusted by the constant term of the QUBO model because
277  // ABS2 Solver does not handle constant terms.
278  if (param.get_target_energy().has_value()) {
279  param.set("target_energy",
280  std::to_string(param.get_target_energy().value() -
281  quad_model.constant()));
282  }
283  return Sol(quad_model,
284  abs2::Solver::operator()(quad_model.get_abs2_model(), param));
