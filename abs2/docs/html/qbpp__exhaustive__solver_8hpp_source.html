250  void flip(vindex_t index) override {
251  vindex_t flip_index = var_order_[index];
252  if (flip_index >= var_count()) {
253  throw std::out_of_range(
254  THROW_MESSAGE("Sol: flip_index (", flip_index, ") out of range"));
255  }
256  energy_ = energy() + delta_[flip_index];
257  for (vindex_t j = 0; j < quad_model_.degree(flip_index); ++j) {
258  auto [k, coeff] = quad_model_.quadratic(flip_index, j);
259  delta_[k] += (2 * get(flip_index) - 1) * (2 * get(k) - 1) * coeff;
260  }
261  delta_[flip_index] = -delta_[flip_index];
262  if (!energy_.has_value()) {
263  energy_ = comp_energy();
264  }
265 
266  bit_vector_.flip(flip_index);
267  }

320 inline void SearchAlgorithm::search() {
321  const int parallel_param = 8;
322  SolDelta sol_delta(*this);
323  if (quad_model_.term_count() == 0) {
324  register_new_sol(sol_delta);
325  return;
326  }
327  if (var_count() <= 16) {
328  register_new_sol(sol_delta);
329  sol_delta.search(var_count() - 1);
330  return;
331  }
332  sol_deltas.push_back(sol_delta);
333  gen_sol_deltas(sol_delta, var_count() - parallel_param);
334 
335  tbb::parallel_for(tbb::blocked_range<size_t>(0, sol_deltas.size()),
336  [&](const tbb::blocked_range<size_t> &range) {
337  for (size_t i = range.begin(); i < range.end(); ++i) {
338  register_new_sol(sol_deltas[i]);
339  sol_deltas[i].search(var_count() -
340  (parallel_param + 1));
341  }
342  });
343 }
