284  void swap_heap(vindex_t i, vindex_t j) {
285  std::swap(heap_[i], heap_[j]);
286  index_[heap_[i].second] = i;
287  index_[heap_[j].second] = j;
288  }
289 
290  void bubble_up(vindex_t i) {
291  while (i > 0) {
292  vindex_t parent = (i - 1) / 2;
293  if (heap_[i] < heap_[parent]) {
294  swap_heap(i, parent);
295  i = parent;
296  } else {
297  break;
298  }
299  }
300  }
301 
302  void bubble_down(vindex_t i) {
303  while (true) {
304  vindex_t left = 2 * i + 1;
305  vindex_t right = 2 * i + 2;
306  vindex_t smallest = i;
307 
308  if (left < heap_size() && heap_[left] < heap_[smallest]) {
309  smallest = left;
310  }
311  if (right < heap_size() && heap_[right] < heap_[smallest]) {
312  smallest = right;
313  }
314  if (smallest != i) {
315  swap_heap(i, smallest);
316  i = smallest;
317  } else {
318  break;
319  }
320  }
321  }
